import pandas as pd
import numpy as np
from sklearn.ensemble import RandomForestClassifier
from sklearn.model_selection import train_test_split

# 1. CARREGAMENTO DAS BASES
base_boletos = pd.read_csv('base_boletos_fiap.csv')
base_aux = pd.read_csv('base_auxiliar_fiap.csv')
graph_results = pd.read_csv('analise_grafos_risco.csv') # Gerado no passo de grafos

# 2. TRATAMENTO E DEFINIÇÃO DO TARGET (DEFAULT)
base_boletos['dt_vencimento'] = pd.to_datetime(base_boletos['dt_vencimento'])
base_boletos['dt_pagamento'] = pd.to_datetime(base_boletos['dt_pagamento'])

# Calculamos o atraso real
base_boletos['atraso'] = (base_boletos['dt_pagamento'] - base_boletos['dt_vencimento']).dt.days

# Definimos Default: Boleto não pago OU atraso maior que 60 dias
base_boletos['is_default'] = ((base_boletos['dt_pagamento'].isnull()) | (base_boletos['atraso'] > 60)).astype(int)

# 3. CRUZAMENTO DE DADOS (MERGE)
df_final = pd.merge(base_boletos, base_aux, left_on='id_pagador', right_on='id_cnpj', how='inner')
df_final = pd.merge(df_final, graph_results[['id_cnpj', 'pagerank', 'out_degree', 'risk_score_raw']], on='id_cnpj', how='inner')

# Seleção de variáveis para o modelo
features = ['vlr_nominal', 'score_materialidade_v2', 'media_atraso_dias', 'pagerank', 'out_degree', 'risk_score_raw']
df_model = df_final.dropna(subset=features + ['is_default'])

# 4. MODELAGEM DA PROBABILIDADE DE DEFAULT (PD)
X = df_model[features]
y = df_model['is_default']

# Treinamos o classificador para obter probabilidades (predict_proba)
clf = RandomForestClassifier(n_estimators=100, random_state=42)
clf.fit(X, y)

# Geramos a coluna PD (Probabilidade de ser 1 - Default)
df_model['probability_of_default'] = clf.predict_proba(X)[:, 1]

# 5. CÁLCULO DA PERDA ESPERADA (EL)
# EL = Valor do Título * Probabilidade de Default
df_model['perda_esperada_el'] = df_model['vlr_nominal'] * df_model['probability_of_default']

# 6. CATEGORIZAÇÃO DE RISCO PARA O GESTOR
def segmentar_risco(pd):
    if pd <= 0.05: return '1. Baixo Risco (0-5%)'
    elif pd <= 0.20: return '2. Risco Moderado (5-20%)'
    elif pd <= 0.50: return '3. Alto Risco (20-50%)'
    else: return '4. Crítico (>50%)'

df_model['categoria_risco'] = df_model['probability_of_default'].apply(segmentar_risco)

# 7. EXPORTAÇÃO PARA EXCEL
colunas_finais = [
    'id_boleto', 'id_pagador', 'vlr_nominal', 'dt_vencimento', 
    'probability_of_default', 'perda_esperada_el', 'categoria_risco'
]
df_model[colunas_finais].to_excel('analise_perda_esperada_el.xlsx', index=False)

print("Planilha de Perda Esperada gerada com sucesso!")
