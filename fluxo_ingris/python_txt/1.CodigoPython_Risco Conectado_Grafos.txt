import pandas as pd
import networkx as nx
import numpy as np

# 1. Carregamento das Bases
base_auxiliar = pd.read_csv('base_auxiliar_fiap.csv')
base_boletos = pd.read_csv('base_boletos_fiap.csv')

# 2. Tratamento de Datas e Cálculo de Atraso
base_boletos['dt_vencimento'] = pd.to_datetime(base_boletos['dt_vencimento'])
base_boletos['dt_pagamento'] = pd.to_datetime(base_boletos['dt_pagamento'])

# Calculamos o atraso em dias (apenas valores positivos representam atraso real)
base_boletos['atraso'] = (base_boletos['dt_pagamento'] - base_boletos['dt_vencimento']).dt.days
base_boletos['atraso_positivo'] = base_boletos['atraso'].apply(lambda x: max(0, x) if pd.notnull(x) else 0)

# 3. Construção do Grafo (Relacionamento Sacado -> Cedente)
G = nx.DiGraph()

# Agrupamos os boletos por par (Pagador/Beneficiário) para consolidar os valores
edges = base_boletos.groupby(['id_pagador', 'id_beneficiario']).agg(
    total_valor=('vlr_nominal', 'sum'),
    qtd_boletos=('id_boleto', 'count'),
    media_atraso_real=('atraso_positivo', 'mean')
).reset_index()

# Adicionamos as conexões (arestas) ao grafo
for _, row in edges.iterrows():
    G.add_edge(row['id_pagador'], row['id_beneficiario'], 
               weight=row['total_valor'], 
               count=row['qtd_boletos'])

# 4. Cálculo de Métricas de Rede
degree_cent = nx.degree_centrality(G)
pagerank = nx.pagerank(G, weight='weight')

# 5. Consolidação dos Dados dos Nós (Empresas)
nodes_data = []
for node in G.nodes():
    # Buscamos informações extras da base auxiliar
    info_aux = base_auxiliar[base_auxiliar['id_cnpj'] == node]
    
    # Criamos um dicionário com os dados consolidados
    node_dict = {
        'id_cnpj': node,
        'in_degree': G.in_degree(node),   # Quantos pagam para ele
        'out_degree': G.out_degree(node), # Para quantos ele deve
        'pagerank': pagerank.get(node, 0),
        'degree_centrality': degree_cent.get(node, 0)
    }
    
    # Se a empresa existir na base auxiliar, trazemos os scores
    if not info_aux.empty:
        node_dict.update({
            'score_materialidade_v2': info_aux['score_materialidade_v2'].values[0],
            'media_atraso_dias_historico': info_aux['media_atraso_dias'].values[0],
            'uf': info_aux['uf'].values[0]
        })
    
    nodes_data.append(node_dict)

df_final = pd.DataFrame(nodes_data)

# 6. Cálculo do Risco de Contágio (Propagação)
# Definimos um score de risco simples: (Atraso médio / 365) + (Inverso do Score de Materialidade)
df_final['risk_score_raw'] = (df_final['media_atraso_dias_historico'].fillna(0) / 365) + \
                             (1 - df_final['score_materialidade_v2'].fillna(500) / 1000)

# Para cada Cedente, calculamos a exposição aos seus pagadores riscosos
contagio_list = []
for node in G.nodes():
    if G.in_degree(node) > 0:
        predecessors = list(G.predecessors(node))
        valor_exposto = 0
        riscos_pagadores = []
        
        for p in predecessors:
            p_risk = df_final.loc[df_final['id_cnpj'] == p, 'risk_score_raw'].values[0]
            riscos_pagadores.append(p_risk)
            # Se o pagador tem risco alto (> 0.5), somamos o valor do boleto à exposição
            if p_risk > 0.5:
                valor_exposto += G[p][node]['weight']
        
        contagio_list.append({
            'id_cnpj': node,
            'avg_payer_risk': np.mean(riscos_pagadores),
            'total_exposed_value': valor_exposto
        })

df_contagio = pd.DataFrame(contagio_list)

# 7. Merge Final e Exportação
df_analise_completa = pd.merge(df_final, df_contagio, on='id_cnpj', how='left')
df_analise_completa.to_excel('analise_grafos_risco.xlsx', index=False)

print("Processamento concluído e base gerada!")
